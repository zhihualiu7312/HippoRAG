Sub GeneratePatentPPT()
    Dim pptPres As Presentation
    Dim slideObj As Slide
    Dim titleShape As Shape, bodyShape As Shape, cardShape As Shape, tableShape As Shape
    Dim tbl As Table
    Dim i As Long, f As Integer, r As Integer, c As Integer
    Dim titles As Variant, descs As Variant
    
    Set pptPres = ActivePresentation
    
    ' 1. 清理已有幻灯片
    For i = pptPres.Slides.Count To 1 Step -1
        pptPres.Slides(i).Delete
    Next i
    
    ' --- Slide 1: 封面 ---
    Set slideObj = pptPres.Slides.Add(1, 12) ' ppLayoutBlank = 12
    slideObj.Background.Fill.Solid
    slideObj.Background.Fill.ForeColor.RGB = RGB(24, 43, 73) ' 科技深蓝
    
    Set titleShape = slideObj.Shapes.AddTextbox(1, 50, 180, 860, 100)
    With titleShape.TextFrame
        .WordWrap = -1
        .TextRange.Text = "高选择性微纳结构制造工艺及其装置" & vbCrLf & "专利可专利性与技术特征答辩"
        .TextRange.Font.Name = "微软雅黑"
        .TextRange.Font.Size = 28
        .TextRange.Font.Bold = -1
        .TextRange.Font.Color.RGB = RGB(255, 255, 255)
    End With
    
    Set bodyShape = slideObj.Shapes.AddTextbox(1, 50, 320, 860, 100)
    With bodyShape.TextFrame
        .WordWrap = -1
        .TextRange.Text = "专利申请号：202610XXXXXX.X  |  申请人：研发中心技术团队" & vbCrLf & "重点答辩方向：针对对比文件 (D1/D2) 的新颖性与非显而易见性证明"
        .TextRange.Font.Name = "微软雅黑"
        .TextRange.Font.Size = 16
        .TextRange.Font.Color.RGB = RGB(180, 205, 235)
    End With
    
    ' --- Slide 2: 现有技术局限与瓶颈 ---
    Set slideObj = pptPres.Slides.Add(2, 12)
    Set titleShape = slideObj.Shapes.AddTextbox(1, 40, 30, 880, 50)
    With titleShape.TextFrame
        .TextRange.Text = "一、现有技术（Prior Art）局限与实质技术问题"
        .TextRange.Font.Name = "微软雅黑"
        .TextRange.Font.Size = 22
        .TextRange.Font.Bold = -1
        .TextRange.Font.Color.RGB = RGB(24, 43, 73)
    End With
    
    ' D1/D2 缺陷卡片
    Set cardShape = slideObj.Shapes.AddShape(5, 40, 100, 420, 380) ' msoShapeRoundedRectangle = 5
    cardShape.Fill.ForeColor.RGB = RGB(245, 247, 250)
    cardShape.Line.Color.RGB = RGB(200, 210, 225)
    With cardShape.TextFrame
        .WordWrap = -1
        .TextRange.Text = "对比文件 D1 / D2 方案缺陷" & vbCrLf & vbCrLf & _
                          "• D1 方案局限：采用常规各向同性工艺，高长径比结构内部反应速率不均，崩塌率 > 15%。" & vbCrLf & vbCrLf & _
                          "• D2 方案局限：虽然引入干法自由基刻蚀，但材料选择比低于 20:1，无法满足亚 3nm 节点精度需求。" & vbCrLf & vbCrLf & _
                          "• 行业共有技术瓶颈：刻蚀副产物积聚导致底渣，且无法兼顾高选择比与低晶格损伤。"
        .TextRange.Font.Name = "微软雅黑"
        .TextRange.Font.Size = 13
        .TextRange.Font.Color.RGB = RGB(45, 55, 72)
        .TextRange.Paragraphs(1).Font.Bold = -1
        .TextRange.Paragraphs(1).Font.Size = 16
        .TextRange.Paragraphs(1).Font.Color.RGB = RGB(24, 43, 73)
    End With
    
    ' 本发明解决的矛盾卡片
    Set cardShape = slideObj.Shapes.AddShape(5, 500, 100, 420, 380)
    cardShape.Fill.ForeColor.RGB = RGB(235, 242, 250)
    cardShape.Line.Color.RGB = RGB(0, 114, 206)
    With cardShape.TextFrame
        .WordWrap = -1
        .TextRange.Text = "本发明旨在解决的实质技术问题" & vbCrLf & vbCrLf & _
                          "1. 物理矛盾：如何在大深宽比狭缝中保持反应物高扩散速率，同时避免侧壁过刻？" & vbCrLf & vbCrLf & _
                          "2. 材料矛盾：如何打破“极低物理损伤”与“超高刻蚀选择比（> 100:1）”不可兼得的传统认知？" & vbCrLf & vbCrLf & _
                          "3. 核心目标：提供兼具原子级精度控制与高产率的各向异性架构与工艺路径。"
        .TextRange.Font.Name = "微软雅黑"
        .TextRange.Font.Size = 13
        .TextRange.Font.Color.RGB = RGB(45, 55, 72)
        .TextRange.Paragraphs(1).Font.Bold = -1
        .TextRange.Paragraphs(1).Font.Size = 16
        .TextRange.Paragraphs(1).Font.Color.RGB = RGB(0, 114, 206)
    End With
    
    ' --- Slide 3: 区别技术特征 ---
    Set slideObj = pptPres.Slides.Add(3, 12)
    Set titleShape = slideObj.Shapes.AddTextbox(1, 40, 30, 880, 50)
    With titleShape.TextFrame
        .TextRange.Text = "二、本发明的核心技术构思与区别特征"
        .TextRange.Font.Name = "微软雅黑"
        .TextRange.Font.Size = 22
        .TextRange.Font.Bold = -1
        .TextRange.Font.Color.RGB = RGB(24, 43, 73)
    End With
    
    titles = Array("区别特征 1：协同气源比例", "区别特征 2：脉冲射频偏置", "区别特征 3：动态原位钝化")
    descs = Array("引入特定三元协同气源，配比控制在 3:1:0.5，在表面形成自限制保护层，不同于 D1 的单一气源。", _
                  "采用双频交替脉冲 RF 能量控制（13.56MHz / 60MHz），将离子轰击能量精准限制在 15eV 以下。", _
                  "在刻蚀周期间隙原位注入单分子保护膜，彻底解决深孔底部的侧向钻蚀问题，D2 未公开此步骤。")
                  
    For f = 0 To 2
        Set cardShape = slideObj.Shapes.AddShape(5, 40 + f * 295, 120, 280, 340)
        cardShape.Fill.ForeColor.RGB = RGB(248, 249, 251)
        cardShape.Line.Color.RGB = RGB(0, 114, 206)
        cardShape.Line.Weight = 1.5
        With cardShape.TextFrame
            .WordWrap = -1
            .TextRange.Text = titles(f) & vbCrLf & vbCrLf & descs(f)
            .TextRange.Font.Name = "微软雅黑"
            .TextRange.Font.Size = 13
            .TextRange.Font.Color.RGB = RGB(45, 55, 72)
            .TextRange.Paragraphs(1).Font.Bold = -1
            .TextRange.Paragraphs(1).Font.Size = 15
            .TextRange.Paragraphs(1).Font.Color.RGB = RGB(0, 114, 206)
        End With
    Next f
    
    ' --- Slide 4: 技术效果与机理 ---
    Set slideObj = pptPres.Slides.Add(4, 12)
    Set titleShape = slideObj.Shapes.AddTextbox(1, 40, 30, 880, 50)
    With titleShape.TextFrame
        .TextRange.Text = "三、技术效果机理与显著进步（证明创造性）"
        .TextRange.Font.Name = "微软雅黑"
        .TextRange.Font.Size = 22
        .TextRange.Font.Bold = -1
        .TextRange.Font.Color.RGB = RGB(24, 43, 73)
    End With
    
    Set cardShape = slideObj.Shapes.AddShape(1, 40, 100, 880, 380) ' msoShapeRectangle = 1
    cardShape.Fill.ForeColor.RGB = RGB(255, 255, 255)
    cardShape.Line.Color.RGB = RGB(210, 215, 225)
    With cardShape.TextFrame
        .WordWrap = -1
        .TextRange.Text = "物理/化学机理与预料不到的技术效果" & vbCrLf & vbCrLf & _
                          "1. 协同机理：特征 1 与特征 2 产生协同反应，活性自由基在特定位点形成 <1nm 超薄动态保护膜，将刻蚀选择比突破性提升至 150:1（较 D1/D2 提升 7.5 倍）。" & vbCrLf & vbCrLf & _
                          "2. 预料不到的效果：界面晶格缺陷密度降低 82%，器件关态漏电流降低一个数量级。本领域技术人员无法根据 D1 和 D2 的常规组合预见该协同效应。" & vbCrLf & vbCrLf & _
                          "3. 实测数据对比：" & vbCrLf & _
                          "   • 材料刻蚀选择比：本发明 150:1  vs  D1 方案 15:1  vs  D2 方案 20:1" & vbCrLf & _
                          "   • 关键尺寸 (CD) 偏差：本发明 < 0.5nm  vs  常规工艺 2.3nm"
        .TextRange.Font.Name = "微软雅黑"
        .TextRange.Font.Size = 13
        .TextRange.Font.Color.RGB = RGB(45, 55, 72)
        .TextRange.Paragraphs(1).Font.Bold = -1
        .TextRange.Paragraphs(1).Font.Size = 16
        .TextRange.Paragraphs(1).Font.Color.RGB = RGB(24, 43, 73)
    End With
    
    ' --- Slide 5: 权利要求映射表 ---
    Set slideObj = pptPres.Slides.Add(5, 12)
    Set titleShape = slideObj.Shapes.AddTextbox(1, 40, 30, 880, 50)
    With titleShape.TextFrame
        .TextRange.Text = "四、独立权利要求与从属权利要求映射表"
        .TextRange.Font.Name = "微软雅黑"
        .TextRange.Font.Size = 22
        .TextRange.Font.Bold = -1
        .TextRange.Font.Color.RGB = RGB(24, 43, 73)
    End With
    
    Set tableShape = slideObj.Shapes.AddTable(4, 3, 40, 100, 880, 350)
    Set tbl = tableShape.Table
    
    tbl.Cell(1, 1).Shape.TextFrame.TextRange.Text = "权利要求项"
    tbl.Cell(1, 2).Shape.TextFrame.TextRange.Text = "核心技术特征定义"
    tbl.Cell(1, 3).Shape.TextFrame.TextRange.Text = "对比文件 (D1/D2) 映射与新颖性/创造性说明"
    
    For r = 1 To 3
        tbl.Cell(1, r).Shape.Fill.ForeColor.RGB = RGB(24, 43, 73)
        tbl.Cell(1, r).Shape.TextFrame.TextRange.Font.Color.RGB = RGB(255, 255, 255)
        tbl.Cell(1, r).Shape.TextFrame.TextRange.Font.Bold = -1
        tbl.Cell(1, r).Shape.TextFrame.TextRange.Font.Size = 13
    Next r
    
    tbl.Cell(2, 1).Shape.TextFrame.TextRange.Text = "权利要求 1 (独权)"
    tbl.Cell(2, 2).Shape.TextFrame.TextRange.Text = "限定三元气源组分及 3:1:0.5 的特定配比参数"
    tbl.Cell(2, 3).Shape.TextFrame.TextRange.Text = "未在 D1 或 D2 中公开。D1 仅揭示单气源体系，不具备三元协同保护机制。"
    
    tbl.Cell(3, 1).Shape.TextFrame.TextRange.Text = "权利要求 2 (从权)"
    tbl.Cell(3, 2).Shape.TextFrame.TextRange.Text = "双频交替脉冲射频（13.56MHz + 60MHz）偏置控制"
    tbl.Cell(3, 3).Shape.TextFrame.TextRange.Text = "D2 采用连续单频射频。本特征有效抑制高能轰击损伤，具备显著创造性。"
    
    tbl.Cell(4, 1).Shape.TextFrame.TextRange.Text = "权利要求 3 (从权)"
    tbl.Cell(4, 2).Shape.TextFrame.TextRange.Text = "原位单分子层保护膜形成与循环解吸步骤"
    tbl.Cell(4, 3).Shape.TextFrame.TextRange.Text = "解决深度各向异性刻蚀中的侧钻问题，说明书实施例 1-3 提供了完整支撑。"
    
    For r = 2 To 4
        For c = 1 To 3
            tbl.Cell(r, c).Shape.Fill.ForeColor.RGB = RGB(248, 250, 253)
            tbl.Cell(r, c).Shape.TextFrame.TextRange.Font.Name = "微软雅黑"
            tbl.Cell(r, c).Shape.TextFrame.TextRange.Font.Size = 12
            tbl.Cell(r, c).Shape.TextFrame.TextRange.Font.Color.RGB = RGB(45, 55, 72)
        Next c
    Next r
    
    MsgBox "专供专利审查评估的演示 PPT 已一键生成完成！", vbInformation, "专利 PPT 生成器"
End Sub
